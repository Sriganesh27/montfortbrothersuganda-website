<?php
// erp/create_admin.php

// 1. Correctly include the config file to get the variables
require_once __DIR__ . '/config/config.php';

// 2. Establish connection using the variables defined in erp/config/config.php
// Note: We use the $variable names as defined in your provided config.php
$conn = new mysqli($DB_HOST, $DB_USER, $DB_PASS, $DB_NAME);

// Check connection
if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

// 3. Define admin details
$adminUsername = 'admin'; 
$adminPassword = 'password123'; // Change this to your desired password
$adminRole = 'Admin';

// Hash the password securely
$hashedPassword = password_hash($adminPassword, PASSWORD_DEFAULT);

try {
    // Check if the username already exists to avoid duplicate errors
    $checkSql = "SELECT id FROM users WHERE username = ?";
    $checkStmt = $conn->prepare($checkSql);
    $checkStmt->bind_param("s", $adminUsername);
    $checkStmt->execute();
    if ($checkStmt->get_result()->num_rows > 0) {
        die("<h3 style='color:orange;'>Notice: User '$adminUsername' already exists.</h3>");
    }

    // 4. Insert the new admin
    // Table structure confirmed from your previous login logic:
    $sql = "INSERT INTO users (username, password, role, is_active, created_at) VALUES (?, ?, ?, 1, NOW())";
    $stmt = $conn->prepare($sql);
    
    if (!$stmt) {
        throw new Exception("Prepare failed: " . $conn->error);
    }

    $stmt->bind_param("sss", $adminUsername, $hashedPassword, $adminRole);

    if ($stmt->execute()) {
        echo "<h3 style='color:green;'>Success! Admin user created.</h3>";
        echo "<b>Username:</b> $adminUsername<br>";
        echo "<b>Password:</b> $adminPassword<br><br>";
        echo "<span style='color:red;'><b>IMPORTANT:</b> Delete this file (create_admin.php) from your server immediately!</span>";
    } else {
        echo "<h3 style='color:red;'>Execution failed:</h3> " . $stmt->error;
    }

    $stmt->close();
} catch (Exception $e) {
    echo "<h3 style='color:red;'>Database Error:</h3> " . $e->getMessage();
}

$conn->close();
?>
